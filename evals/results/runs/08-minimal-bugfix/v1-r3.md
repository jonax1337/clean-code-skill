Customers with no orders trigger two empty-array edge cases (G3):

- `average([])` computes `0 / 0`, which is `NaN`.
- `Math.max()` with no arguments returns `-Infinity`.

I fixed both at the source, so `average` is also safe for other callers. I chose `0` as the value for "no orders", so `avg` shows `"0.00"` and `max` shows `0`. If the dashboard should show something else, such as `"–"`, change those two spots.

```js
// stats.js
function average(values) {
  if (values.length === 0) return 0;
  let sum = 0;
  for (const v of values) sum += v;
  return sum / values.length;
}

function maximum(values) {
  return values.length === 0 ? 0 : Math.max(...values);
}

function summarize(orders) {
  const amounts = orders.map(o => o.amount);
  return {
    count: orders.length,
    avg: average(amounts).toFixed(2),
    max: maximum(amounts),
  };
}

module.exports = { average, summarize };
```
