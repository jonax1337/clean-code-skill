The cause is that an empty order list hits two edge cases: `0 / 0` gives `NaN` in `average`, and `Math.max()` with no arguments gives `-Infinity`. I fixed both at the source, so `average` is also safe for other callers.

```js
// stats.js
const NO_VALUE = 0;

function average(values) {
  if (values.length === 0) return NO_VALUE;
  return sum(values) / values.length;
}

function maximum(values) {
  if (values.length === 0) return NO_VALUE;
  return Math.max(...values);
}

function sum(values) {
  let total = 0;
  for (const v of values) total += v;
  return total;
}

function summarize(orders) {
  const amounts = orders.map(o => o.amount);
  return {
    count: orders.length,
    avg: average(amounts).toFixed(2),
    max: maximum(amounts),
  };
}

module.exports = { average, summarize };
```
