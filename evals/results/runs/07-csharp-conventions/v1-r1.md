```csharp
using Microsoft.EntityFrameworkCore;

public interface ICustomerRepository
{
    Task<Customer?> GetByIdAsync(Guid id, CancellationToken ct);
    Task<Customer?> GetByEmailAsync(string email, CancellationToken ct);
    Task AddAsync(Customer customer, CancellationToken ct);
}

public sealed class SqlCustomerRepository : ICustomerRepository
{
    private readonly DbContext _db;

    public SqlCustomerRepository(DbContext db) => _db = db;

    public Task<Customer?> GetByIdAsync(Guid id, CancellationToken ct) =>
        _db.Set<Customer>().FindAsync(new object[] { id }, ct).AsTask();

    public Task<Customer?> GetByEmailAsync(string email, CancellationToken ct) =>
        _db.Set<Customer>().FirstOrDefaultAsync(customer => customer.Email == email, ct);

    public async Task AddAsync(Customer customer, CancellationToken ct)
    {
        _db.Add(customer);
        await _db.SaveChangesAsync(ct);
    }
}
```

Notes:

- **Naming (N-rules, "ein Wort pro Konzept"):** I named the lookup `GetByEmailAsync`, not `FindByEmailAsync`, so it matches `GetByIdAsync`. Mixing `Find` and `Get` for the same concept would be inconsistent.
- **Null return:** `Customer?` mirrors the existing `Order?` contract (G11, team convention wins). The skill prefers Special-Case objects or exceptions over null, but changing that would be a cross-cutting decision for both repositories.
- **Assumption:** `Customer.Email` is a `string`. If it's a value object, the predicate needs adjusting. Email case sensitivity depends on your DB collation. If it's case-sensitive, normalize the email on write and lookup, or add a unique index on a normalized column.
- **Duplication (G5):** The two repositories now share the same `GetByIdAsync`/`AddAsync` shape. With only two, I kept them separate. A third would be the point to extract a generic base.
- I haven't compiled or tested this. A test for `GetByEmailAsync` covering a match, no match, and a different-case email would be the first one to add.
